==================== README.txt ====================

النسخة النهائية لمشروع تطوير تطبيقات الويب

تم تطبيق المتطلبات الأساسية:
1- واجهة مستخدم حديثة ومتجاوبة.
2- تسجيل دخول.
3- إنشاء أول حساب مدير بشكل آمن.
4- Dashboard.
5- CRUD كامل للكورسات: إضافة/عرض/تعديل/حذف.
6- CRUD كامل للطلاب: تسجيل/عرض/تعديل/حذف.
7- Validation للمدخلات.
8- Password hashing و password_verify.
9- Prepared Statements لمنع SQL Injection.
10- htmlspecialchars عند عرض البيانات لمنع XSS.
11- صلاحيات admin.
12- CSRF على عمليات POST.
13- رفع الصور مع فحص MIME فعلي وحد 5MB.
14- Transaction عند ربط الطالب بالكورس وعند حذفه.
15- قاعدة البيانات والجداول موجودة في schema.sql.

التشغيل:
- استورد schema.sql في phpMyAdmin.
- تأكد من db.php.
- ضع المشروع داخل htdocs.
- افتح register_admin.php مرة واحدة لإنشاء المدير الأول.
- بعدها افتح login.php.
- بعد تسجيل الدخول تصبح index.php هي Dashboard.



==================== schema.sql ====================

CREATE DATABASE IF NOT EXISTS courses_db CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE courses_db;
CREATE TABLE IF NOT EXISTS admins(id INT AUTO_INCREMENT PRIMARY KEY,name VARCHAR(100) NOT NULL,email VARCHAR(190) NOT NULL UNIQUE,password VARCHAR(255) NOT NULL,role VARCHAR(30) NOT NULL DEFAULT 'user',status TINYINT(1) NOT NULL DEFAULT 1,last_login_at DATETIME NULL,created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP) ENGINE=InnoDB;
CREATE TABLE IF NOT EXISTS course(id INT AUTO_INCREMENT PRIMARY KEY,course_name VARCHAR(150) NOT NULL,price DECIMAL(10,2) NOT NULL DEFAULT 0,course_image VARCHAR(255) NOT NULL) ENGINE=InnoDB;
CREATE TABLE IF NOT EXISTS students(id INT AUTO_INCREMENT PRIMARY KEY,student_name VARCHAR(150) NOT NULL,phone VARCHAR(30) DEFAULT '',student_image VARCHAR(255) NOT NULL,created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP) ENGINE=InnoDB;
CREATE TABLE IF NOT EXISTS course_student(course_id INT NOT NULL,student_id INT NOT NULL,PRIMARY KEY(course_id,student_id),FOREIGN KEY(course_id) REFERENCES course(id) ON DELETE CASCADE,FOREIGN KEY(student_id) REFERENCES students(id) ON DELETE CASCADE) ENGINE=InnoDB;

function verify_csrf() {
    if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
        http_response_code(405);
        die("طريقة الطلب غير مسموحة.");
    }
    if (empty($_POST['csrf_token']) || empty($_SESSION['csrf_token'])) {
        http_response_code(403);
        die("طلب غير صالح.");
    }
    if (!hash_equals($_SESSION['csrf_token'], $_POST['csrf_token'])) {
        http_response_code(403);
        die("انتهت صلاحية الطلب أو أن الطلب غير صالح.");
    }
}
